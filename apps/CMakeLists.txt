file(GLOB FIBER_APP_CHILDREN RELATIVE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/*")
list(SORT FIBER_APP_CHILDREN)

foreach(FIBER_APP_DIR IN LISTS FIBER_APP_CHILDREN)
    set(FIBER_APP_PATH "${CMAKE_CURRENT_LIST_DIR}/${FIBER_APP_DIR}")
    if (NOT IS_DIRECTORY "${FIBER_APP_PATH}")
        continue()
    endif()
    if (FIBER_APP_DIR MATCHES "^_")
        continue()
    endif()
    if (EXISTS "${FIBER_APP_PATH}/CMakeLists.txt")
        add_subdirectory("${FIBER_APP_DIR}" "${CMAKE_BINARY_DIR}/apps-build/${FIBER_APP_DIR}")
    endif()
endforeach()
